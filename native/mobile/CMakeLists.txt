# Copy only platform adaptations into the build tree. Upstream remains pristine.

function(mobile_patch name)
  file(READ "${PEERCAST_ROOT}/${name}" content)
  if(name STREQUAL "core/common/matroska.h")
    string(REPLACE "#include <string>" "#include <string>\n#include \"byte_traits.h\"" content "${content}")
    string(REPLACE "std::basic_string<uint8_t>" "std::basic_string<uint8_t, MobileByteTraits>" content "${content}")
    string(REPLACE "bytes += b;" "bytes.push_back(b);" content "${content}")
    string(REPLACE "bytes += is.readChar();" "bytes.push_back(is.readChar());" content "${content}")
  elseif(name STREQUAL "core/unix/usocket.cpp" OR name STREQUAL "core/common/sslclientsocket.cpp")
    set(old "stats.add(Stats::BYTESOUT, r);")
    string(FIND "${content}" "${old}" match)
    if(match EQUAL -1)
      message(FATAL_ERROR "Upstream socket accounting changed; review mobile patch")
    endif()
    string(PREPEND content "#include \"external_traffic.h\"\n")
    string(REPLACE "${old}" "mobileExternalTraffic.recordSent(r, host.localIP());\n        ${old}" content "${content}")
  elseif(name STREQUAL "core/common/sys.h")
    string(REPLACE "#if _BIG_ENDIAN" "#if defined(__BYTE_ORDER__) && __BYTE_ORDER__ == __ORDER_BIG_ENDIAN__" content "${content}")
    string(REPLACE "#ifdef __BIG_ENDIAN__" "#if defined(__BYTE_ORDER__) && __BYTE_ORDER__ == __ORDER_BIG_ENDIAN__" content "${content}")
  elseif(name MATCHES "core/common/cookie")
    string(REPLACE "operator ==(const Cookie &c)" "operator ==(const Cookie &c) const" content "${content}")
  elseif(name STREQUAL "core/common/servent.cpp")
    # Normalize only localhost HTTP playback copies, including replacement headers.
    string(REPLACE "#include \"servent.h\"" "#include \"servent.h\"\n#include \"flv_playback_header.h\"" content "${content}")
    string(REPLACE "ch->headPack.writeRaw(bsock);" "ChanPacket playbackHead;\n            playbackHead = ch->headPack;\n            if (sock->host.isLocalhost()) mobileRepairFlvHeader(playbackHead.data, playbackHead.len);\n            playbackHead.writeRaw(bsock);" content "${content}")
    string(REPLACE "rawPack.writeRaw(bsock);" "if (rawPack.type == ChanPacket::T_HEAD && sock->host.isLocalhost())\n                                mobileRepairFlvHeader(rawPack.data, rawPack.len);\n                            rawPack.writeRaw(bsock);" content "${content}")
    string(REPLACE "if (!isPrivate())\n    {\n        if  (servMgr->bitrateFull" "if (type == T_RELAY || !isPrivate())\n    {\n        if  (servMgr->bitrateFull" content "${content}")
  elseif(name STREQUAL "core/common/channel.cpp")
    # HTTP 503 is translated into PCP QUIT/UNAVAILABLE (1003) by readStream.
    # Keep a busy tracker eligible for the existing MIN_TRACKER_RETRY backoff.
    # Other failures and ordinary relay peers retain their dead-hit handling.
    set(old "if (!ch->sourceHost.tracker || ((error != 503) && ch->sourceHost.tracker))")
    string(FIND "${content}" "${old}" match)
    if(match EQUAL -1)
      message(FATAL_ERROR "Upstream tracker retry condition changed; review mobile patch")
    endif()
    string(REPLACE "${old}" "if (!ch->sourceHost.tracker || (error != 503 && error != PCP_ERROR_QUIT + PCP_ERROR_UNAVAILABLE))" content "${content}")
  elseif(name STREQUAL "core/common/servmgr.h")
    string(REPLACE "numStreams(Servent::T_RELAY, false) >= maxRelays" "numStreams(Servent::T_RELAY, true) >= maxRelays" content "${content}")
  elseif(name STREQUAL "core/common/servhs.cpp")
    string(REPLACE "void Servent::handshakeHTTP(HTTP &http, bool isHTTP)\n{" "extern bool mobileRequestAllowed(const char*, bool);\nvoid Servent::handshakeHTTP(HTTP &http, bool isHTTP)\n{\n    if (!mobileRequestAllowed(http.cmdLine, sock->host.isLocalhost()))\n        throw HTTPException(HTTP_SC_FORBIDDEN, 403);" content "${content}")
  elseif(name STREQUAL "core/unix/usys.cpp")
    string(REPLACE "#ifdef __APPLE__" "#if 0" content "${content}")
    string(REPLACE "#ifdef _GNU_SOURCE" "#if 0" content "${content}")
    string(REPLACE "#ifndef __APPLE__" "#if 1" content "${content}")
    string(REPLACE "signal(SIGABRT, SIG_IGN);" "// Do not alter the host application's SIGABRT handler." content "${content}")
    if(APPLE)
      string(REPLACE "retval = system(cmdLine.c_str());"
        "retval = -1; // iOS cannot launch shell commands."
        content "${content}")
    endif()
  elseif(name STREQUAL "core/common/servmgr.cpp")
    string(REPLACE "servMgr->channelDirectory->update();" "// Directory retrieval is owned by Dart." content "${content}")
    string(REPLACE "servMgr->uptestServiceRegistry->update();" "// No external upload test service on mobile." content "${content}")
  endif()

  file(WRITE "${PEERCAST_ROOT}/${name}" "${content}")

endfunction()

if(APPLE)
  # Apple's SDK includes <_string.h>. Search the copied PeerCast headers only
  # after SDK headers so its own _string.h does not shadow the libc header.
  target_compile_options(peercast_mobile PRIVATE
    "-idirafter${PEERCAST_ROOT}/core/common"
    "-idirafter${PEERCAST_ROOT}/core/unix")
else()
  target_include_directories(peercast_mobile PRIVATE
    "${PEERCAST_ROOT}/core/common" "${PEERCAST_ROOT}/core/unix")
endif()
target_compile_definitions(peercast_mobile PRIVATE _UNIX PEERCAST_MOBILE)
target_link_libraries(peercast_mobile PRIVATE ssl crypto)
if(ANDROID)
  target_link_libraries(peercast_mobile PRIVATE log)
  target_link_options(peercast_mobile PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_LIST_DIR}/exports.map" "-Wl,-z,max-page-size=16384")
elseif(APPLE)
  target_link_options(peercast_mobile PRIVATE "-Wl,-exported_symbols_list,${CMAKE_CURRENT_LIST_DIR}/exports.apple")
endif()
